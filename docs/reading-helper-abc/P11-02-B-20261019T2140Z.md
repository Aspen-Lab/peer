/** The text between the parentheses of each `useLayoutEffect(` / `useEffect(` call in `source`, with whether it has a dependency list. */
export function effectsOf(source: string): Array<{ hook: string; body: string; deps: boolean }> {
  const out: Array<{ hook: string; body: string; deps: boolean }> = [];
  const call = /\b(useLayoutEffect|useEffect)\(/g;
  for (let match = call.exec(source); match; match = call.exec(source)) {
    let depth = 1;
    let i = match.index + match[0].length;
    for (; i < source.length && depth > 0; i += 1) {
      if (source[i] === "(") depth += 1;
      else if (source[i] === ")") depth -= 1;
    }
    const body = source.slice(match.index + match[0].length, i - 1);
    // Drop trailing comments / whitespace; a call whose last argument is the arrow's block has no list (`}`), one with a list ends `]`.
    const tail = body.replace(/\/\/[^\n]*\n?/g, "").trimEnd().replace(/,$/, "").trimEnd();
    out.push({ hook: match[1], body, deps: tail.endsWith("]") });
  }
  return out;
}

/** What is wrong with the effects that run after every commit: a state write that is not `setCardHeight` behind `settleHeight`. */
export function everyCommitFaults(source: string): string[] {
  const faults: string[] = [];
  for (const effect of effectsOf(source)) {
    if (effect.deps) continue;
    const writes = [...effect.body.matchAll(/\b(set[A-Z]\w*)\(/g)].map((m) => m[1]);
    for (const write of new Set(writes)) {
      if (write === "setCardHeight" && /\bsettleHeight\(/.test(effect.body)) continue;
      faults.push(`${effect.hook} with no dependency list calls ${write}`);
    }
  }
  return faults;
}

### 3.4a Does P11-01's three-pane layout remove the mechanism?

**It removes the occasion, not the class, unless it deletes the measured-height state.** What the mechanism needs is a state written from a layout read after every commit, read back by the layout (card height -> `max-height` / `top` -> card height). In a margin pane:

- `reserveDock`, `--explain-dock`, `--explain-dock-width`, `DOCK_RESERVE_PX` and the P7-15 reflow effect go (the pane takes the width itself): that removes the page-width feedback and the `viewport.width >= 1024` coupling (`docked`). Note `reserveDock` is also used by `lib/reader/windows.ts` (stickies, the notes window: P10-09d) and `notes-window.tsx`; P11-01 must move those in the same change or the stack loses its reserve.
- The card's height **can** be pure CSS (a pane that scrolls; the card `max-height` as a `calc` of the pane's height less the card's top) with the card's top written from the selection's y in the **frame-coalesced watcher** (scroll / resize / the Fit subscription), never from an effect after every commit. Then `cardHeight`, `settleHeight`, the guard and the every-commit effect are all deleted, and the loop class has no place to stand. **Recommended**, but only if the invariant 3.3.1 (the whole window in view) is kept by CSS (`max-height` against the pane's visible height).
- If P11-01 keeps "lift the card until its bottom is in view" it needs the measured height again: then **keep `settleHeight` and the new pin as they are** and do not read the pane's size in a no-list effect.
- **A new surface P11-01 creates: the dragged boundaries** (item 2). A drag changes the pane's width on every pointer move; the text reflows; the selection's rectangle moves; the watcher's `setMeasured` writes. That is a feedback chain with a human in it, and a `ResizeObserver` callback that writes a size which changes the observed element's own size is the same class as the loop (and a browser's own "ResizeObserver loop" warning). Rules for the C: write the pane width from the pointer handler only (one state, rounded to whole pixels, coalesced to a frame); never observe the element whose size you write; compare with 1 px tolerance before setting; keep the box's commit count per drag frame at one (run `h/flow.mjs`'s instrument over a drag and over the collapse of the left pane; the longest burst must stay under 10).
